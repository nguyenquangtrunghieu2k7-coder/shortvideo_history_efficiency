# -*- coding: utf-8 -*-
from pathlib import Path
import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
DATA_DIR = ROOT / "data" / "processed"

EVAL_EVENTS_FILE = DATA_DIR / "evaluation_events.parquet"
EVAL_HISTORY_FILE = DATA_DIR / "evaluation_history.parquet"

SPLIT_FILES = {
    "TRAIN": DATA_DIR / "train.parquet",
    "VALIDATION": DATA_DIR / "validation.parquet",
    "TEST": DATA_DIR / "test.parquet",
}

EXPECTED_EVAL_ROWS = 43_027
EXPECTED_SPLIT_ROWS = {
    "TRAIN": 30_118,
    "VALIDATION": 6_454,
    "TEST": 6_455,
}

EVAL_EVENT_REQUIRED = [
    "evaluation_id",
    "user_id",
    "video_id",
    "time_ms",
    "long_view",
]

HISTORY_REQUIRED = [
    "eval_idx",
    "user_id",
    "video_id",
    "time_ms",
    "history_len_full",
    "history_1",
    "history_1_len",
    "history_5",
    "history_5_len",
    "history_10",
    "history_10_len",
    "history_20",
    "history_20_len",
    "history_50",
    "history_50_len",
    "history_100",
    "history_100_len",
    "history_full",
    "history_full_len",
]

WINDOWS = [1, 5, 10, 20, 50, 100]


def check(condition, message):
    if condition:
        print(f"[PASS] {message}")
    else:
        print(f"[FAIL] {message}")
        raise AssertionError(message)


def check_file(path):
    check(path.exists(), f"File exists: {path}")


def check_columns(df, required, name):
    missing = [c for c in required if c not in df.columns]
    check(
        not missing,
        f"{name} contains required columns"
        + (f" | missing: {missing}" if missing else ""),
    )


def check_no_columns(df, forbidden, name):
    present = [c for c in forbidden if c in df.columns]
    check(
        not present,
        f"{name} does not contain forbidden columns"
        + (f" | found: {present}" if present else ""),
    )


def audit_evaluation_events():
    print("=" * 80)
    print("1. EVALUATION EVENT ARTIFACT")
    print("=" * 80)
    
    check_file(EVAL_EVENTS_FILE)
    df = pd.read_parquet(EVAL_EVENTS_FILE)

    print(f"  rows: {len(df):,}")
    print(f"  columns: {len(df.columns)}")

    check(
        len(df) == EXPECTED_EVAL_ROWS,
        f"evaluation_events rows = {EXPECTED_EVAL_ROWS:,}",
    )

    check_columns(
        df,
        EVAL_EVENT_REQUIRED,
        "evaluation_events",
    )

    check(
        "evaluation_id" in df.columns,
        "evaluation_events uses evaluation_id",
    )

    check(
        "eval_idx" not in df.columns,
        "evaluation_events does NOT incorrectly require eval_idx",
    )

    check(
        df["evaluation_id"].is_unique,
        "evaluation_id is unique",
    )

    expected_ids = pd.RangeIndex(len(df))
    actual_ids = pd.Index(df["evaluation_id"].astype("int64"))

    check(
        actual_ids.equals(expected_ids),
        "evaluation_id covers 0..N-1",
    )

    unique_target = df["long_view"].dropna().unique()

    check(
        set(unique_target).issubset({0, 1}),
        "long_view is binary",
    )

    check(
        not df[["user_id", "video_id", "time_ms"]].duplicated().any(),
        "evaluation event key (user_id, video_id, time_ms) is unique",
    )

    return df


def audit_evaluation_history(eval_events):
    print()
    print("=" * 80)
    print("2. EVALUATION HISTORY ARTIFACT")
    print("=" * 80)
    
    check_file(EVAL_HISTORY_FILE)
    history = pd.read_parquet(EVAL_HISTORY_FILE)

    print(f"  rows: {len(history):,}")
    print(f"  columns: {len(history.columns)}")

    check(
        len(history) == EXPECTED_EVAL_ROWS,
        f"evaluation_history rows = {EXPECTED_EVAL_ROWS:,}",
    )

    check_columns(
        history,
        HISTORY_REQUIRED,
        "evaluation_history",
    )

    check(
        "eval_idx" in history.columns,
        "evaluation_history uses eval_idx",
    )

    check(
        "evaluation_id" not in history.columns,
        "evaluation_history does NOT incorrectly require evaluation_id",
    )

    check(
        history["eval_idx"].is_unique,
        "eval_idx is unique in evaluation_history",
    )

    expected_idx = pd.RangeIndex(len(history))
    actual_idx = pd.Index(history["eval_idx"].astype("int64"))

    check(
        actual_idx.equals(expected_idx),
        "eval_idx covers 0..N-1",
    )

    event_ids = set(eval_events["evaluation_id"].astype("int64"))
    history_ids = set(history["eval_idx"].astype("int64"))

    check(
        history_ids.issubset(event_ids),
        "every history row maps to an evaluation event",
    )

    check(
        event_ids.issubset(history_ids),
        "every evaluation event has a corresponding history row",
    )

    event_key = (
        eval_events[["evaluation_id", "user_id", "video_id", "time_ms"]]
        .set_index("evaluation_id")
        .sort_index()
    )

    history_key = (
        history[["eval_idx", "user_id", "video_id", "time_ms"]]
        .rename(columns={"eval_idx": "evaluation_id"})
        .set_index("evaluation_id")
        .sort_index()
    )

    check(
        not history[["user_id", "video_id", "time_ms"]].duplicated().any(),
        "history event key (user_id, video_id, time_ms) is unique",
    )

    check(
        event_key.index.equals(history_key.index),
        "evaluation event and history ID domains match exactly",
    )

    check(
        event_key[["user_id", "video_id", "time_ms"]].equals(
            history_key[["user_id", "video_id", "time_ms"]]
        ),
        "evaluation event and history content keys match exactly",
    )

    # History artifact must not contain the target label.
    check_no_columns(
        history,
        ["long_view"],
        "evaluation_history",
    )

    for w in WINDOWS:
        len_col = f"history_{w}_len"
        history_col = f"history_{w}"

        check(
            (history[len_col] >= 0).all(),
            f"{len_col} contains no negative values",
        )

        check(
            (history[len_col] <= w).all(),
            f"{len_col} never exceeds W={w}",
        )

        check(
            history[history_col].apply(len).equals(history[len_col]),
            f"{history_col} length matches {len_col}",
        )

    for w in WINDOWS:
        check(
            (history["history_full_len"] >= history[f"history_{w}_len"]).all(),
            f"history_full_len >= history_{w}_len for every row",
        )

    check(
        history["history_full"].apply(len).equals(history["history_full_len"]),
        "history_full length matches history_full_len",
    )

    return history


def audit_splits(eval_events, eval_history):
    print()
    print("=" * 80)
    print("3. TRAIN / VALIDATION / TEST SPLITS")
    print("=" * 80)
    
    all_split_indices = []

    for split_name, path in SPLIT_FILES.items():
        print()
        print("-" * 80)
        print(split_name)

        check_file(path)
        df = pd.read_parquet(path)

        print(f"  rows: {len(df):,}")
        print(f"  columns: {len(df.columns)}")

        expected_rows = EXPECTED_SPLIT_ROWS[split_name]

        check(
            len(df) == expected_rows,
            f"{split_name.lower()} rows = {expected_rows:,}",
        )

        # Split artifacts are derived from evaluation_history.
        check_columns(
            df,
            HISTORY_REQUIRED,
            f"{split_name.lower()}",
        )

        check(
            "eval_idx" in df.columns,
            f"{split_name.lower()} uses eval_idx",
        )

        # These are intentionally absent.
        check_no_columns(
            df,
            ["evaluation_id", "long_view"],
            f"{split_name.lower()}",
        )

        check(
            df["eval_idx"].is_unique,
            f"{split_name.lower()} eval_idx is unique",
        )

        split_idx = set(df["eval_idx"].astype("int64"))
        history_idx = set(eval_history["eval_idx"].astype("int64"))

        check(
            split_idx.issubset(history_idx),
            f"{split_name.lower()} eval_idx values exist in evaluation_history",
        )

        all_split_indices.append((split_name, split_idx))

    # Splits must be mutually exclusive.
    for i in range(len(all_split_indices)):
        name_a, idx_a = all_split_indices[i]

        for j in range(i + 1, len(all_split_indices)):
            name_b, idx_b = all_split_indices[j]

            check(
                idx_a.isdisjoint(idx_b),
                f"{name_a.lower()} and {name_b.lower()} are mutually exclusive",
            )

    train_idx = all_split_indices[0][1]
    val_idx = all_split_indices[1][1]
    test_idx = all_split_indices[2][1]

    union = train_idx | val_idx | test_idx
    all_eval_idx = set(eval_history["eval_idx"].astype("int64"))

    check(
        union == all_eval_idx,
        "train + validation + test cover all evaluation_history rows exactly once",
    )

    # Verify that labels are recoverable through eval_idx -> evaluation_id.
    event_by_id = eval_events.set_index("evaluation_id")

    for split_name, split_idx in all_split_indices:
        missing = [
            idx for idx in split_idx if idx not in event_by_id.index
        ]

        check(
            not missing,
            f"{split_name.lower()} eval_idx values map to evaluation_events",
        )

        labels = event_by_id.loc[list(split_idx), "long_view"]

        check(
            labels.notna().all(),
            f"{split_name.lower()} labels are available through evaluation_events",
        )


def main():
    print("=" * 80)
    print("PHASE 8 — SCHEMA & SPLIT MAPPING AUDIT")
    print("=" * 80)
    
    eval_events = audit_evaluation_events()
    eval_history = audit_evaluation_history(eval_events)
    audit_splits(eval_events, eval_history)

    print()
    print("=" * 80)
    print("PHASE 8 SCHEMA AUDIT: PASS")
    print("=" * 80)


if __name__ == "__main__":
    main()

from pathlib import Path

import pandas as pd


ROOT = Path(__file__).resolve().parents[2]

INPUT_PATH = ROOT / "data" / "processed" / "evaluation_history.parquet"
OUTPUT_DIR = ROOT / "data" / "processed"

TRAIN_RATIO = 0.70
VAL_RATIO = 0.15


def main():
    print("=" * 80)
    print("BUILD TEMPORAL TRAIN / VALIDATION / TEST SPLIT")
    print("=" * 80)

    print(f"\nLoading:")
    print(f"  {INPUT_PATH}")

    if not INPUT_PATH.exists():
        raise FileNotFoundError(INPUT_PATH)

    df = pd.read_parquet(INPUT_PATH)

    # Strict chronological ordering.
    df = df.sort_values("time_ms").reset_index(drop=True)

    n = len(df)

    train_end = int(n * TRAIN_RATIO)
    val_end = int(n * (TRAIN_RATIO + VAL_RATIO))

    train = df.iloc[:train_end].copy()
    val = df.iloc[train_end:val_end].copy()
    test = df.iloc[val_end:].copy()

    splits = {
        "train": train,
        "validation": val,
        "test": test,
    }

    print("\nSplit sizes:")

    for name, split in splits.items():
        print(
            f"  {name:<10}: "
            f"{len(split):>7,} events | "
            f"{split['time_ms'].min()} → "
            f"{split['time_ms'].max()}"
        )

    # Validate temporal boundaries before writing.
    assert train["time_ms"].max() < val["time_ms"].min()
    assert val["time_ms"].max() < test["time_ms"].min()

    # Validate no events were lost.
    assert len(train) + len(val) + len(test) == len(df)

    # Validate eval_idx uniqueness.
    assert train["eval_idx"].is_unique
    assert val["eval_idx"].is_unique
    assert test["eval_idx"].is_unique

    # Validate split disjointness by evaluation event.
    train_idx = set(train["eval_idx"])
    val_idx = set(val["eval_idx"])
    test_idx = set(test["eval_idx"])

    assert not (train_idx & val_idx)
    assert not (train_idx & test_idx)
    assert not (val_idx & test_idx)

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

    paths = {
        "train": OUTPUT_DIR / "train.parquet",
        "validation": OUTPUT_DIR / "validation.parquet",
        "test": OUTPUT_DIR / "test.parquet",
    }

    for name, split in splits.items():
        split.to_parquet(paths[name], index=False)
        print(f"\nSaved {name}:")
        print(f"  {paths[name]}")

    print("\n" + "=" * 80)
    print("SPLIT VALIDATION")
    print("=" * 80)

    print("Temporal ordering: PASS")
    print("Event coverage:    PASS")
    print("eval_idx unique:   PASS")
    print("Split disjointness: PASS")

    print("\n" + "=" * 80)
    print("DONE")
    print("=" * 80)


if __name__ == "__main__":
    main()

from pathlib import Path

import pandas as pd


ROOT = Path(__file__).resolve().parents[2]

HISTORY_PATH = ROOT / "data" / "processed" / "evaluation_history.parquet"

TRAIN_RATIO = 0.70
VAL_RATIO = 0.15
TEST_RATIO = 0.15

RANDOM_STATE = 42


def build_user_split(df):
    users = df["user_id"].drop_duplicates().sample(
        frac=1.0,
        random_state=RANDOM_STATE,
    )

    n = len(users)

    n_train = int(n * TRAIN_RATIO)
    n_val = int(n * VAL_RATIO)

    train_users = set(users.iloc[:n_train])
    val_users = set(users.iloc[n_train:n_train + n_val])
    test_users = set(users.iloc[n_train + n_val:])

    return train_users, val_users, test_users


def audit_split(df, train_users, val_users, test_users):
    print("\n" + "=" * 80)
    print("USER-LEVEL SPLIT AUDIT")
    print("=" * 80)

    sets = {
        "train": train_users,
        "validation": val_users,
        "test": test_users,
    }

    print("Users:")
    for name, users in sets.items():
        print(f"  {name:<10}: {len(users):,}")

    print("\nUser overlap:")

    failed = False

    names = list(sets.keys())

    for i in range(len(names)):
        for j in range(i + 1, len(names)):
            a = names[i]
            b = names[j]

            overlap = sets[a] & sets[b]

            print(f"  {a} ∩ {b}: {len(overlap):,}")

            if overlap:
                failed = True

    print("\nEvaluation events:")

    for name, users in sets.items():
        subset = df[df["user_id"].isin(users)]

        print(
            f"  {name:<10}: "
            f"{len(subset):>7,} events "
            f"({len(subset) / len(df):.2%})"
        )

    if failed:
        print("\nUser-disjoint split: FAIL")
    else:
        print("\nUser-disjoint split: PASS")


def audit_event_distribution(df, train_users, val_users, test_users):
    print("\n" + "=" * 80)
    print("SPLIT DISTRIBUTION AUDIT")
    print("=" * 80)

    splits = {
        "train": train_users,
        "validation": val_users,
        "test": test_users,
    }

    for name, users in splits.items():
        subset = df[df["user_id"].isin(users)]

        click_rate = subset["is_click"].mean() \
            if "is_click" in subset.columns else None

        print(f"\n{name}:")
        print(f"  users:  {subset['user_id'].nunique():,}")
        print(f"  events: {len(subset):,}")

        if click_rate is not None:
            print(f"  click rate: {click_rate:.2%}")


def audit_user_history_leakage(df, train_users, val_users, test_users):
    print("\n" + "=" * 80)
    print("USER HISTORY LEAKAGE AUDIT")
    print("=" * 80)

    all_users = set(df["user_id"])

    assigned = train_users | val_users | test_users

    missing = all_users - assigned
    extra = assigned - all_users

    print(f"Users missing from split: {len(missing):,}")
    print(f"Unknown users in split:   {len(extra):,}")

    if not missing and not extra:
        print("User assignment coverage: PASS")
    else:
        print("User assignment coverage: FAIL")


def main():
    print("=" * 80)
    print("AUDIT TRAIN / VALIDATION / TEST SPLIT")
    print("=" * 80)

    print(f"\nLoading:")
    print(f"  {HISTORY_PATH}")

    if not HISTORY_PATH.exists():
        raise FileNotFoundError(HISTORY_PATH)

    df = pd.read_parquet(HISTORY_PATH)

    print(f"\nEvaluation events: {len(df):,}")
    print(f"Unique users:      {df['user_id'].nunique():,}")

    train_users, val_users, test_users = build_user_split(df)

    audit_split(
        df,
        train_users,
        val_users,
        test_users,
    )

    audit_user_history_leakage(
        df,
        train_users,
        val_users,
        test_users,
    )

    print("\n" + "=" * 80)
    print("AUDIT COMPLETE")
    print("=" * 80)


if __name__ == "__main__":
    main()

from pathlib import Path

import pandas as pd


ROOT = Path(__file__).resolve().parents[2]

HISTORY_PATH = ROOT / "data" / "processed" / "evaluation_history.parquet"

TRAIN_RATIO = 0.70
VAL_RATIO = 0.15
TEST_RATIO = 0.15


def audit_global_time_order(df):
    print("\n" + "=" * 80)
    print("GLOBAL TEMPORAL ORDER")
    print("=" * 80)

    df = df.sort_values("time_ms").reset_index(drop=True)

    n = len(df)

    train_end = int(n * TRAIN_RATIO)
    val_end = int(n * (TRAIN_RATIO + VAL_RATIO))

    train = df.iloc[:train_end]
    val = df.iloc[train_end:val_end]
    test = df.iloc[val_end:]

    print("Events:")
    print(f"  train:      {len(train):,}")
    print(f"  validation: {len(val):,}")
    print(f"  test:       {len(test):,}")

    print("\nTime ranges:")

    for name, split in [
        ("train", train),
        ("validation", val),
        ("test", test),
    ]:
        print(
            f"  {name:<10} "
            f"{split['time_ms'].min()} "
            f"→ {split['time_ms'].max()}"
        )

    train_val_ok = train["time_ms"].max() < val["time_ms"].min()
    val_test_ok = val["time_ms"].max() < test["time_ms"].min()

    print("\nTemporal ordering:")

    print(
        f"  train < validation: "
        f"{'PASS' if train_val_ok else 'FAIL'}"
    )

    print(
        f"  validation < test:  "
        f"{'PASS' if val_test_ok else 'FAIL'}"
    )


def audit_user_temporal_order(df):
    print("\n" + "=" * 80)
    print("PER-USER TEMPORAL ORDER")
    print("=" * 80)

    ordered = df.sort_values(
        ["user_id", "time_ms"]
    )

    violations = (
        ordered
        .groupby("user_id")["time_ms"]
        .diff()
        .lt(0)
        .sum()
    )

    print(f"Per-user time violations: {violations:,}")

    if violations == 0:
        print("Per-user temporal order: PASS")
    else:
        print("Per-user temporal order: FAIL")


def audit_history_before_evaluation(df):
    print("\n" + "=" * 80)
    print("HISTORY BEFORE EVALUATION")
    print("=" * 80)

    # history.py already validated temporal construction.
    # Here we verify that every evaluation event has a valid
    # non-negative history length.

    violations = 0

    for w in [1, 5, 10, 20, 50, 100]:
        col = f"history_{w}_len"

        violations += (
            (df[col] < 0) |
            (df[col] > w)
        ).sum()

    print(f"History window violations: {violations:,}")

    if violations == 0:
        print("History bounds: PASS")
    else:
        print("History bounds: FAIL")


def audit_user_presence_across_temporal_splits(df):
    print("\n" + "=" * 80)
    print("USER PRESENCE ACROSS TEMPORAL SPLITS")
    print("=" * 80)

    df = df.sort_values("time_ms")

    n = len(df)

    train_end = int(n * TRAIN_RATIO)
    val_end = int(n * (TRAIN_RATIO + VAL_RATIO))

    train = df.iloc[:train_end]
    val = df.iloc[train_end:val_end]
    test = df.iloc[val_end:]

    train_users = set(train["user_id"])
    val_users = set(val["user_id"])
    test_users = set(test["user_id"])

    print(f"Train users:      {len(train_users):,}")
    print(f"Validation users: {len(val_users):,}")
    print(f"Test users:       {len(test_users):,}")

    print("\nOverlaps:")

    print(f"  train ∩ validation: {len(train_users & val_users):,}")
    print(f"  train ∩ test:       {len(train_users & test_users):,}")
    print(f"  validation ∩ test:  {len(val_users & test_users):,}")


def main():
    print("=" * 80)
    print("AUDIT TEMPORAL TRAIN / VALIDATION / TEST SPLIT")
    print("=" * 80)

    print(f"\nLoading:")
    print(f"  {HISTORY_PATH}")

    if not HISTORY_PATH.exists():
        raise FileNotFoundError(HISTORY_PATH)

    df = pd.read_parquet(HISTORY_PATH)

    print(f"\nEvaluation events: {len(df):,}")

    audit_global_time_order(df)
    audit_user_temporal_order(df)
    audit_history_before_evaluation(df)
    audit_user_presence_across_temporal_splits(df)

    print("\n" + "=" * 80)
    print("AUDIT COMPLETE")
    print("=" * 80)


if __name__ == "__main__":
    main()
